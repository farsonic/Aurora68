# Aurora68 (PiStorm-SE) - config.txt for the Raspberry Pi firmware.
# Lines beginning with # are comments. The firmware reads this file at power-on.

[all]
kernel=kernel8.img            # Aurora68 (the Emu68 fork, build variant psmac)
arm_64bit=1
# at a fixed address, not followkernel: a container with a disk image inside is 100 MB and more, and
# behind the kernel it reaches into the kernel's early tables (the first Pi 3B boot got a corrupted
# image and the blinking question mark; docs/08 item 35)
initramfs aurora68.cpio 0x10000000   # the container: Macintosh IIci ROM, declaration ROM, firmware
                                       # (make it with make-container.py, see READ-ME-FIRST.txt)
# (no disable_splash: the firmware's rainbow square for a moment at power-on says the firmware
#  found its files; with it suppressed a card the firmware rejects and a dead board look the same)
avoid_warnings=1
enable_uart=1                 # the kernel's log on the serial pins (115200 8N1); leave on
dtoverlay=disable-bt          # the kernel writes its log on the PL011, which this puts on GPIO 14/15
                              # (the header's TXD/RXD); Bluetooth, which shares that UART on a Pi 3,
                              # stays off (bt=off in cmdline.txt) until the kernel moves it

# The Pi's HDMI output is the Macintosh's second screen (640 x 480, 256 colours). The
# firmware sets the HDMI mode at power-on and the kernel shows the Mac's frame buffer in it;
# 640 x 480 at 60 Hz maps one pixel to one. A larger mode is scaled by the firmware.
hdmi_force_hotplug=1
hdmi_group=2                  # DMT (monitor) modes
hdmi_mode=4                   # 640 x 480, 60 Hz. 9 = 800 x 600, 16 = 1024 x 768, 82 = 1080p
disable_overscan=1

[pi3]
gpu_mem=64
# A Pi 3B carries the BCM43438 WiFi (firmware brcmfmac43430-sdio.bin, .txt and .clm_blob); a 3B+ or
# 3A+ the BCM43455 (cyfmac43455-sdio-standard.bin and friends). The kernel picks by chip id; put the
# right trio in the container (READ-ME-FIRST.txt).
# A Pi 3B+ or 3A+ carries the BCM43455 WiFi (firmware cyfmac43455-sdio-standard.bin and friends);
# a Pi 3B the BCM43438 (brcmfmac43430-sdio.bin and brcmfmac43430-sdio.txt). The kernel picks by
# chip id. Bluetooth shares the Pi 3's one PL011 with the console; see docs/12 in the repository.

[pi4]
gpu_mem=64
