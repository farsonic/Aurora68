# Aurora68 - config.txt for the Raspberry Pi firmware.
# Lines beginning with # are comments. The firmware reads this file at power-on.

[all]
kernel=kernel8.img            # Aurora68 (the Emu68 fork, build variant psmac)
arm_64bit=1
# at a fixed address, not followkernel: a container with a disk image inside is 100 MB and more, and
# behind the kernel it reaches into the kernel's early tables (the first Pi 3B boot got a corrupted
# image and the blinking question mark; docs/08 item 35)
initramfs aurora68.cpio 0x10000000   # the container: Macintosh IIci ROM, declaration ROM, firmware
                                       # (make it with make-container.py, see READ-ME-FIRST.txt)
boot_delay=0                  # start the kernel at once (some firmware versions wait a second by default)
# Booting fast (READ-ME-FIRST.txt, "Booting fast"): the firmware reads the whole container before the
# kernel starts, 10 to 20 MB a second on a Pi 3, so keep disk images out of it and put them on the
# card beside it (MyMac.hda): the kernel reads them while the Mac starts.
# (no disable_splash: the firmware's rainbow square for a moment at power-on says the firmware
#  found its files; with it suppressed a card the firmware rejects and a dead board look the same.
#  disable_splash=1 saves a little more time once the card is known to work.)
avoid_warnings=1
enable_uart=1                 # the kernel's log on the serial pins (115200 8N1); leave on
# no dtoverlay=disable-bt: Bluetooth has the PL011 (GPIO 30-33) and the kernel moves its log to the
# mini UART on GPIO 14/15 the moment Bluetooth starts (same pins, same 115200 8N1)

# The Pi's HDMI output is the Macintosh's second screen, at the size of the HDMI mode below. The
# firmware sets the HDMI mode at power-on and the kernel shows the Mac's frame buffer in it, one
# pixel to one (until 20 November 2026 a mode larger than the declaration ROM's size was scaled).
hdmi_force_hotplug=1
hdmi_group=2                  # DMT (monitor) modes
hdmi_mode=4                   # 640 x 480, 60 Hz. 9 = 800 x 600, 16 = 1024 x 768, 82 = 1080p,
                              # 69 = 1920 x 1200 (60 Hz, reduced blanking; 68 is too fast for a Pi 3)
# The Mac's screen takes the size of this mode (the kernel fits the declaration ROM to it at every
# start; mac.screen=WxH in cmdline.txt only to differ from it), so any declaration ROM in the container
# serves every mode: aurora68-se.declrom for the SE's two screens. Every size up to 2560 x 1440 offers
# millions of colours in Monitors once 32-bit addressing is on; gpu_mem=64 below is enough for it.
# 2560 x 1440 has no mode a Pi 3 can drive at 60 Hz (241 MHz of pixel clock): use the custom mode at
# 30 Hz with reduced blanking (about 121 MHz), with these five lines in place of hdmi_mode= above:
#   hdmi_group=2
#   hdmi_mode=87
#   hdmi_cvt=2560 1440 30 3 0 0 1
#   max_framebuffer_width=2560
#   max_framebuffer_height=1440
disable_overscan=1
hdmi_drive=2                  # HDMI, not DVI: the Mac's sound goes to the monitor's loudspeakers (mac.audio=hdmi,
                              # the default). The firmware otherwise picks DVI for many monitors, and DVI has no sound

[pi3]
gpu_mem=64
# In the SE, with the HDMI screen at 640 x 480 or none, gpu_mem=16 makes the firmware use its
# smaller cut-down start_cd.elf (copy start_cd.elf and fixup_cd.dat to the card first, or the Pi will
# not start): less to read at power-on. Not for the larger HDMI screens, whose frame buffer needs
# the 64 MB. Not measured on hardware yet.
# [pi3] is the 3B, 3B+ and 3A+ (the 3A+ boots with the 3B+'s bcm2710-rpi-3-b-plus.dtb). A Pi 3B
# carries the BCM43438 WiFi (cyfmac43430-sdio.bin, brcmfmac43430-sdio.txt, .clm_blob); a 3B+ or 3A+
# the BCM43455 (cyfmac43455-sdio-standard.bin and friends). The kernel picks by chip id; put the
# right set in the container (READ-ME-FIRST.txt; get-firmware.sh fetches every board's).
# Bluetooth shares the Pi 3's one PL011 with the console; see docs/12 in the repository.
# USB keyboards and mice on the Pi 3's ports are driven by the kernel (usb=off in cmdline.txt stops it).

[pi02]
# The Zero 2 W is not a [pi3] to the firmware: it reads [pi0] and [pi02]. The same settings as the
# Pi 3 (start.elf, fixup.dat, bcm2710-rpi-zero-2-w.dtb; mini-HDMI). Its WiFi is a BCM43436 or
# 43436s (brcmfmac43436-sdio.* or brcmfmac43436s-sdio.*), its Bluetooth BCM43430B0.hcd.
gpu_mem=64

[pi4]
gpu_mem=64
# The Pi 4's Gigabit Ethernet and WiFi are driven by the kernel (the Mac sees both cards). Its USB
# ports (the VL805's xHCI) are not yet in the Macintosh build: Bluetooth keyboards and mice work.
# The same card boots a Pi 3 and a Pi 4: start4.elf, fixup4.dat and bcm2711-rpi-4-b.dtb beside the
# Pi 3's files (get-firmware.sh fetches them all).

[all]
# PiDisplay (the Mac program; docs/10, "PiDisplay") chooses the HDMI mode: the kernel writes it into
# aurora68-display.txt beside this file (the previous one as aurora68-display.bak) and never edits this
# file. Keep the include line last: the firmware takes the last value of a setting, so an hdmi_ line
# after it would win over PiDisplay's. The release's aurora68-display.txt holds comments only, so the
# hdmi_ lines above apply until PiDisplay writes one.
include aurora68-display.txt
